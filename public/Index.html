const express = require('express'); const path = require('path'); const { Pool } = require('pg'); const app = express(); const PORT = process.env.PORT || 3000; app.use(express.json({ limit: '10mb' })); const pool = process.env.DATABASE_URL ? new Pool({ connectionString: process.env.DATABASE_URL, ssl: process.env.DATABASE_URL.includes('localhost') ? false : { rejectUnauthorized: false } }) : null; let databaseReady = false; async function initializeDatabase() { if (!pool) { console.warn('DATABASE_URL não configurada.'); return; } await pool.query(` CREATE TABLE IF NOT EXISTS estoque_app_state ( id INTEGER PRIMARY KEY, state JSONB NOT NULL, updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ) `); databaseReady = true; console.log('Tabela de estado do estoque pronta.'); } app.get('/api/health', (req, res) => { res.json({ ok: true, service: 'Estoque de Aviamentos' }); }); app.get('/api/db-status', async (req, res) => { if (!pool) { return res.status(503).json({ configured: false, connected: false, message: 'DATABASE_URL não configurada.' }); } try { await pool.query('SELECT 1'); res.json({ configured: true, connected: true, message: 'Conexão com PostgreSQL funcionando.' }); } catch (error) { console.error('Erro de conexão:', error.message); res.status(503).json({ configured: true, connected: false, message: 'Não foi possível conectar ao PostgreSQL.' }); } }); app.get('/api/state', async (req, res) => { if (!pool || !databaseReady) { return res.status(503).json({ error: 'Banco de dados indisponível.' }); } try { const result = await pool.query( 'SELECT state FROM estoque_app_state WHERE id = 1' ); if (result.rows.length === 0) { return res.json({ empty: true, state: null }); } res.json({ empty: false, state: result.rows[0].state }); } catch (error) { console.error('Erro ao carregar estado:', error.message); res.status(500).json({ error: 'Falha ao carregar os dados.' }); } }); app.put('/api/state', async (req, res) => { if (!pool || !databaseReady) { return res.status(503).json({ error: 'Banco de dados indisponível.' }); } const state = req.body; if ( !state || typeof state !== 'object' || !Array.isArray(state.products) || !Array.isArray(state.moves) || !Array.isArray(state.cuts) ) { return res.status(400).json({ error: 'Formato de dados inválido.' }); } try { await pool.query( `INSERT INTO estoque_app_state (id, state, updated_at) VALUES (1, $1::jsonb, NOW()) ON CONFLICT (id) DO UPDATE SET state = EXCLUDED.state, updated_at = NOW()`, [JSON.stringify(state)] ); res.json({ ok: true, message: 'Dados salvos no PostgreSQL.' }); } catch (error) { console.error('Erro ao salvar estado:', error.message); res.status(500).json({ error: 'Falha ao salvar os dados.' }); } }); const publicPath = path.join(__dirname, 'public'); app.use('/public', express.static(publicPath)); app.use(express.static(__dirname)); app.get('/', (req, res) => { res.sendFile(path.join(__dirname, 'index.html')); }); async function start() { try { if (pool) { await initializeDatabase(); } app.listen(PORT, '0.0.0.0', () => { console.log(`Servidor iniciado na porta ${PORT}`); }); } catch (error) { console.error('Erro ao inicializar o banco:', error.message); process.exit(1); } } start(); process.on('SIGTERM', async () => { if (pool) await pool.end(); process.exit(0); });
